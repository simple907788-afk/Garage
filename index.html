  <!-- =========================
       JavaScript
       ========================= -->

  <script>

    /* -------------------------
       요소 가져오기
       ------------------------- */

    const target =
      document.querySelector("#garageTarget");

    const status =
      document.querySelector("#status");

    const playButton =
      document.querySelector("#playButton");

    const restartButton =
      document.querySelector("#restartButton");

    const music =
      document.querySelector("#music");


    /* -------------------------
       이미지 인식됨
       ------------------------- */

    target.addEventListener(
      "targetFound",
      function () {

        console.log(
          "GARAGE BAND TARGET FOUND"
        );

        status.innerText =
          "GARAGE BAND — TRACK 01";

        playButton.style.display =
          "inline-block";

        restartButton.style.display =
          "none";
      }
    );


    /* -------------------------
       이미지가 카메라에서 사라짐
       ------------------------- */

    target.addEventListener(
      "targetLost",
      function () {

        console.log(
          "GARAGE BAND TARGET LOST"
        );

        status.innerText =
          "AIM AT THE GARAGE";

        /*
          음악은 여기서 자동으로
          정지시키지 않는다.

          따라서 티셔츠를 잠깐 놓쳐도
          음악은 계속 재생된다.
        */

      }
    );


    /* -------------------------
       PLAY
       ------------------------- */

    playButton.addEventListener(
      "click",
      async function () {

        try {

          await music.play();

          status.innerText =
            "NOW PLAYING — TRACK 01";

          playButton.style.display =
            "none";

          restartButton.style.display =
            "inline-block";

        }

        catch (error) {

          console.log(
            "Audio playback error:",
            error
          );

          status.innerText =
            "TAP PLAY TO START";

        }

      }
    );


    /* -------------------------
       REPLAY
       ------------------------- */

    restartButton.addEventListener(
      "click",
      async function () {

        try {

          music.currentTime = 0;

          await music.play();

          status.innerText =
            "NOW PLAYING — TRACK 01";

        }

        catch (error) {

          console.log(
            "Replay error:",
            error
          );

        }

      }
    );


    /* -------------------------
       음악이 끝났을 때
       ------------------------- */

    music.addEventListener(
      "ended",
      function () {

        status.innerText =
          "TRACK 01 — ENDED";

        restartButton.style.display =
          "inline-block";

      }
    );


  </script>
